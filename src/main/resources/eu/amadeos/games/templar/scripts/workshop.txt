    set act1

  end
end

  if found-page
    say maria psalter.look.after
  else
    say maria psalter.look.1

# --8<-- [start:bonefolder]
on use damp-brush psalter
  if found-page
    say maria psalter.use.after
    stop
  end
  anim maria reach
  lose damp-brush
  gain brush
  set softened
  say maria psalter.damp
  set goal 12
end

on use bonefolder
  anim maria reach
  hide bonefolder
  gain bonefolder
  say maria bonefolder.take
end

on use bonefolder psalter
  if found-page
    say maria psalter.use.after
  else
    if not softened
      anim maria reach
      say maria bonefolder.dry
      set goal 11
    else
      anim maria reach
      sound paper
      show flap
      say maria discover.1
      sound discovery
      anim maria reach
      gain page
      set found-page
      say maria discover.2
      say maria discover.3
      set goal 13
    end
  end
end
# --8<-- [end:bonefolder]

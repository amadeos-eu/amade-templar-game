# Act 3, scene 8: the archive in the tower. The other half of the chirograph, a locked door, and a way out.

on enter archive
  if not been-archive
    set been-archive
    say ines-archive archive.intro.1
    say ines-archive archive.intro.2
    say maria archive.intro.3
    say ines-archive archive.intro.4
    walk ines-archive 40 318
    hide ines-archive
    sound door
    set goal 100
  end
end

on exit archive-door
  if rescued
    goto cloister arrival
    stop
  end
  if locked
    say maria archive-door.locked
    stop
  end
  if not joined
    say maria archive-door.notyet
    stop
  end
  goto charola from-archive
end

on look registers
  say maria registers.look.1
  say maria registers.look.2
end

on use registers
  say maria registers.use
end

# ---- the register on the lectern ----

on look register
  if found-half
    say maria register.empty
    stop
  end
  say maria register.look.1
  say maria register.look.2
  set goal 101
end

on use register
  if found-half
    say maria register.empty
    stop
  end
  anim maria reach
  sound paper
  say maria register.use.1
  say maria register.use.2
  set goal 101
end

on use bonefolder register
  if found-half
    say maria register.empty
    stop
  end
  anim maria reach
  sound paper
  wait 0.6
  sound discovery
  gain half
  set found-half
  say maria half.found.1
  say maria half.found.2
  set goal 102
end

on use page half
  if joined
    stop
  end
  anim maria point
  sound discovery
  lose page
  lose half
  gain chirograph
  set joined
  say maria joined.1
  say maria joined.2
  say maria joined.3
  say maria joined.4
  say maria joined.5
  wait 0.4
  show helene-archive
  sound door
  say helene-archive joined.helene.1
  say maria joined.helene.2
  say helene-archive joined.helene.3
  say maria joined.helene.4
  say helene-archive joined.helene.5
  hide helene-archive
  sound door
  wait 0.4
  say maria joined.helene.6
  set locked
  set goal 103
end

on use half page
  if joined
    stop
  end
  say maria half.page
end

# ---- the book lift ----

on look book-lift
  if lift-up
    say maria book-lift.up
  else
    say maria book-lift.look
  end
end

on use book-lift
  if lift-up
    if locked
      say maria book-lift.small
    else
      say maria book-lift.up
    end
    stop
  end
  anim maria reach
  sound creak
  wait 0.8
  set lift-up
  show lift-box
  show lunch-bag

  say maria book-lift.use.2
end

# ---- the window over the cloister ----

on look tower-window
  say maria tower-window.look.1
  if locked
    say maria tower-window.look.2
  end
end

on use tower-window
  if not locked
    say maria tower-window.notyet
    stop
  end
  anim maria reach
  sound creak
  hide tower-window
  show tower-window-open
  set window-open
  say maria tower-window.open.1
  say maria tower-window.open.2
  set goal 104
end

on look tower-window-open
  say maria tower-window.look.1
  say maria tower-window.look.2
end

on use tower-window-open
  say maria tower-window.shout
end

on use pastry tower-window
  say maria pastry.closed
end

on use pastry tower-window-open
  anim maria reach
  lose pastry
  wait 0.6
  show pigeon-archive
  sound coo
  set pigeon-here
  say maria pastry.window.1
  say maria pastry.window.2
  say maria pastry.window.3
  if has note
    set goal 106
  else
    set goal 105
  end
end

on use note tower-window
  say maria note.closed
end

on use note tower-window-open
  if not pigeon-here
    say maria note.nobody
    stop
  end
  anim maria reach
  say maria rescue.1
  sound camera
  wait 0.8
  say maria rescue.2
  say maria rescue.3
  wait 1.0
  sound door
  show ines-archive
  lose note
  say ines-archive rescue.4
  say ines-archive rescue.5
  say maria rescue.6
  say ines-archive rescue.7
  set rescued
  goto cloister arrival
end

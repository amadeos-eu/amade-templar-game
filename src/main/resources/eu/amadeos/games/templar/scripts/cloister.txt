    set been-cloister
    show ines-cloister
    say helene-cloister cloister.intro.1
    say lena cloister.intro.2
    say helene-cloister cloister.intro.3
    say ines-cloister cloister.intro.4

    wait 0.6
    say helene-cloister cloister.intro.5
    say helene-cloister cloister.intro.6
    say kev-cloister cloister.intro.7
    set goal 110
  end
end

# The proof: the account and the chest tell the same story.

on use chirograph helene-cloister
  if not seen-bills
    say lena finale.early
    stop
  end
  anim lena point
  say lena finale.1
  say lena finale.2
  anim helene-cloister reach
  sound paper
  say helene-cloister finale.3
  say helene-cloister finale.4
  sound discovery
  say lena finale.5
  say kev-cloister finale.6
  say ines-cloister finale.7
  say helene-cloister finale.8
  say helene-cloister finale.9
  walk helene-cloister 20 340
  hide helene-cloister
  sound door
  wait 0.5
  say lena finale.10
  say ines-cloister finale.11
  say lena finale.12
  say lena finale.13
  anim kev-cloister point
  say kev-cloister finale.14
  say lena finale.15
  say kev-cloister finale.16
  sound coo
  say lena finale.17
  title card.finale
  set finished
end

# Act 2, scene 5: the old harbour of La Rochelle at dusk. Kev, an old fisherman, and the weather.

on enter harbour
  if not been-harbour
    set been-harbour
    if kept-phone
      gain phone
    end
    if kept-page
      gain page
    end
    if kept-bonefolder
      gain bonefolder
    end
    if kept-brush
      gain brush
    end
    title card.rochelle
    say lena harbour.intro.1
    say lena harbour.intro.2
    face kev-harbour right
    say kev-harbour harbour.kev.1
    say lena harbour.kev.2
    say kev-harbour harbour.kev.3
    say lena harbour.kev.4
    set goal 60
  end
end

on look sunset
  say lena sunset.look
end

on look saint-nicolas
  say lena saint-nicolas.look.1
  say lena saint-nicolas.look.2
end

on use saint-nicolas
  say lena saint-nicolas.use
end

on look chain-tower
  say lena chain-tower.look
end

on use chain-tower
  say lena chain-tower.use
end

on look harbour-water
  say lena harbour-water.look
end

on use harbour-water
  say lena harbour-water.use
end

on look boat
  say lena boat.look
end

on use boat
  say lena boat.use
end

on look nets
  say lena nets.look
end

on use nets
  say lena nets.use
end

on look gulls
  say lena gulls.look
end

on use gulls
  say lena gulls.use
end

on use sandwich gulls
  say lena gulls.sandwich
end

on look clock-gate
  say lena clock-gate.look
end

on use clock-gate
  say lena clock-gate.use
end

on look office
  say lena office.look
end

on use office
  say lena office.use
end

on look barometer
  if read-barometer
    say lena barometer.again
    stop
  end
  say lena barometer.dark
end

on use barometer
  if read-barometer
    say lena barometer.again
    stop
  end
  say lena barometer.dark
end

on use phone barometer
  if read-barometer
    say lena barometer.again
    stop
  end
  anim lena phone
  sound camera
  say lena barometer.read.1
  say lena barometer.read.2
  set read-barometer
  if fisher-weather
    set can-forecast
    set goal 62
  end
end

on use phone office
  say lena office.phone
end

on look quay-lamp
  say lena quay-lamp.look
end

on look quay-bollard
  say lena quay-bollard.look
end

on use quay-bollard
  say lena quay-bollard.use
end

on exit old-town
  if not fisher-done
    say lena old-town.where
    stop
  end
  goto court arrival
end

# ---- Kev ----

on look kev-harbour
  say lena kev-harbour.look
end

on talk kev-harbour
  dialog kev-harbour
end

option kev-harbour who kev-harbour.q.who once
  say lena kev-harbour.who.1
  say kev-harbour kev-harbour.who.2
  say kev-harbour kev-harbour.who.3
  say lena kev-harbour.who.4
end

option kev-harbour fleet kev-harbour.q.fleet once
  say lena kev-harbour.fleet.1
  say kev-harbour kev-harbour.fleet.2
  say lena kev-harbour.fleet.3
  say kev-harbour kev-harbour.fleet.4
end

option kev-harbour weather kev-harbour.q.weather once if fisher-weather
  say lena kev-harbour.weather.1
  anim kev-harbour point
  say kev-harbour kev-harbour.weather.2
  say lena kev-harbour.weather.3
  say kev-harbour kev-harbour.weather.4
  say lena kev-harbour.weather.5
  say kev-harbour kev-harbour.weather.6
  set new-rochelle
end

option kev-harbour bye kev-harbour.q.bye
  say kev-harbour kev-harbour.bye
  exit
end

on use page kev-harbour
  say lena kev-harbour.page
end

# ---- the old fisherman ----

on look fisherman
  say lena fisherman.look
end

on talk fisherman
  dialog fisherman
end

option fisherman templars fisherman.q.templars if not fisher-weather
  say lena fisherman.templars.1
  say fisherman fisherman.templars.2
  say lena fisherman.templars.3
  say fisherman fisherman.templars.4
  say fisherman fisherman.templars.5
  set fisher-weather
  if read-barometer
    set can-forecast
    set goal 62
  else
    set goal 61
  end
end

option fisherman again fisherman.q.again if fisher-weather
  if fisher-done
    say lena fisherman.done.1
    say fisherman fisherman.done.2
    stop
  end
  say lena fisherman.again.1
  say fisherman fisherman.again.2
end

option fisherman app fisherman.q.app once if new-rochelle
  say lena fisherman.app.1
  say fisherman fisherman.app.2
  say fisherman fisherman.app.3
end

option fisherman barometer fisherman.q.barometer once if can-forecast
  say lena fisherman.barometer.1
  say fisherman fisherman.barometer.2
  say fisherman fisherman.barometer.3
  say lena fisherman.barometer.4
  say fisherman fisherman.barometer.5
  say fisherman fisherman.barometer.6
  say fisherman fisherman.barometer.7
  set fisher-done
  set goal 63
end

option fisherman towers fisherman.q.towers once
  say lena fisherman.towers.1
  say fisherman fisherman.towers.2
end

option fisherman bye fisherman.q.bye
  say fisherman fisherman.bye
  exit
end

on use page fisherman
  say lena fisherman.page
end

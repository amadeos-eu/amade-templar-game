# Act 2, scene 5: the old harbour of La Rochelle at dusk. Kev, an old fisherman, and the weather.

on enter harbour
  if not been-harbour
    set been-harbour
    if kept-phone
      gain phone
    end
    if kept-page
      gain page
    end
    if kept-bonefolder
      gain bonefolder
    end
    if kept-brush
      gain brush
    end
    title card.rochelle
    say maria harbour.intro.1
    say maria harbour.intro.2
    face kev-harbour right
    say kev-harbour harbour.kev.1
    say maria harbour.kev.2
    say kev-harbour harbour.kev.3
    say maria harbour.kev.4
    set goal 60
  end
end

on look sunset
  say maria sunset.look
end

on look saint-nicolas
  say maria saint-nicolas.look.1
  say maria saint-nicolas.look.2
end

on use saint-nicolas
  say maria saint-nicolas.use
end

on look chain-tower
  say maria chain-tower.look
end

on use chain-tower
  say maria chain-tower.use
end

on look harbour-water
  say maria harbour-water.look
end

on use harbour-water
  say maria harbour-water.use
end

on look boat
  say maria boat.look
end

on use boat
  say maria boat.use
end

on look nets
  say maria nets.look
end

on use nets
  say maria nets.use
end

on look gulls
  say maria gulls.look
end

on use gulls
  say maria gulls.use
end

on use sandwich gulls
  say maria gulls.sandwich
end

on look clock-gate
  say maria clock-gate.look
end

on use clock-gate
  say maria clock-gate.use
end

on look office
  say maria office.look
end

on use office
  say maria office.use
end

on look barometer
  if read-barometer
    say maria barometer.again
    stop
  end
  say maria barometer.dark
end

on use barometer
  if read-barometer
    say maria barometer.again
    stop
  end
  say maria barometer.dark
end

on use phone barometer
  if read-barometer
    say maria barometer.again
    stop
  end
  anim maria phone
  sound camera
  say maria barometer.read.1
  say maria barometer.read.2
  set read-barometer
  if fisher-weather
    set can-forecast
    set goal 62
  end
end

on use phone office
  say maria office.phone
end

on look quay-lamp
  say maria quay-lamp.look
end

on look quay-bollard
  say maria quay-bollard.look
end

on use quay-bollard
  say maria quay-bollard.use
end

on exit old-town
  if not fisher-done
    say maria old-town.where
    stop
  end
  goto court arrival
end

# ---- Kev ----

on look kev-harbour
  say maria kev-harbour.look
end

on talk kev-harbour
  dialog kev-harbour
end

option kev-harbour who kev-harbour.q.who once
  say maria kev-harbour.who.1
  say kev-harbour kev-harbour.who.2
  say kev-harbour kev-harbour.who.3
  say maria kev-harbour.who.4
end

option kev-harbour fleet kev-harbour.q.fleet once
  say maria kev-harbour.fleet.1
  say kev-harbour kev-harbour.fleet.2
  say maria kev-harbour.fleet.3
  say kev-harbour kev-harbour.fleet.4
end

option kev-harbour weather kev-harbour.q.weather once if fisher-weather
  say maria kev-harbour.weather.1
  anim kev-harbour point
  say kev-harbour kev-harbour.weather.2
  say maria kev-harbour.weather.3
  say kev-harbour kev-harbour.weather.4
  say maria kev-harbour.weather.5
  say kev-harbour kev-harbour.weather.6
  set new-rochelle
end

option kev-harbour bye kev-harbour.q.bye
  say kev-harbour kev-harbour.bye
  exit
end

on use page kev-harbour
  say maria kev-harbour.page
end

# ---- the old fisherman ----

on look fisherman
  say maria fisherman.look
end

on talk fisherman
  dialog fisherman
end

option fisherman templars fisherman.q.templars if not fisher-weather
  say maria fisherman.templars.1
  say fisherman fisherman.templars.2
  say maria fisherman.templars.3
  say fisherman fisherman.templars.4
  say fisherman fisherman.templars.5
  set fisher-weather
  if read-barometer
    set can-forecast
    set goal 62
  else
    set goal 61
  end
end

option fisherman again fisherman.q.again if fisher-weather
  if fisher-done
    say maria fisherman.done.1
    say fisherman fisherman.done.2
    stop
  end
  say maria fisherman.again.1
  say fisherman fisherman.again.2
end

option fisherman app fisherman.q.app once if new-rochelle
  say maria fisherman.app.1
  say fisherman fisherman.app.2
  say fisherman fisherman.app.3
end

option fisherman barometer fisherman.q.barometer once if can-forecast
  say maria fisherman.barometer.1
  say fisherman fisherman.barometer.2
  say fisherman fisherman.barometer.3
  say maria fisherman.barometer.4
  say fisherman fisherman.barometer.5
  say fisherman fisherman.barometer.6
  say fisherman fisherman.barometer.7
  set fisher-done
  set goal 63
end

option fisherman towers fisherman.q.towers once
  say maria fisherman.towers.1
  say fisherman fisherman.towers.2
end

option fisherman bye fisherman.q.bye
  say fisherman fisherman.bye
  exit
end

on use page fisherman
  say maria fisherman.page
end

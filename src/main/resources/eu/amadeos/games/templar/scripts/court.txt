# Act 2, scene 6: the Cour du Temple at night, where the Templars of La Rochelle had their house.

on enter court
  if fog-seen
    if not act2-done
      set act2-done
    if kept-phone
      gain phone
    end
    if kept-page
      gain page
    end
    if kept-bonefolder
      gain bonefolder
    end
    if kept-brush
      gain brush
    end
    if kept-napkin
      gain napkin
    end
    if kept-token
      gain token
    end
      say maria court.end.1
      say maria court.end.2
      say maria court.end.3
      anim kev-court point
      say kev-court court.end.4
      say maria court.end.5
      say kev-court court.end.6
      say maria court.end.7
      title card.act2.end
      goto charola arrival
    end
    stop
  end
  if not been-court
    set been-court
    say maria court.intro.1
    say kev-court court.intro.2
    say maria court.intro.3
    set goal 70
  end
end

on look night-sky
  say maria night-sky.look
end

on look old-wall
  say maria old-wall.look
end

on use old-wall
  say maria old-wall.use
end

on look street-sign
  say maria street-sign.look
end

on look cafe
  say maria cafe.look
end

on use cafe
  say maria cafe.use
end

on look cafe-window
  say maria cafe-window.look
end

on look terrace-lights
  say maria terrace-lights.look
end

on exit back-harbour
  if act2-done
    stop
  end
  goto harbour old-town
end

# ---- the menu board and the stone behind it ----

on look chalkboard
  if decoded
    say maria chalkboard.decoded
  else
    say maria chalkboard.look.1
    say maria chalkboard.look.2
  end
  say maria chalkboard.behind
end

on use chalkboard
  if not allowed
    say maria chalkboard.ask
    stop
  end
  anim maria reach
  sound door
  hide chalkboard
  show moved-board
  set board-moved
  say maria chalkboard.moved
  set goal 72
end

on use napkin chalkboard
  if decoded
    say maria chalkboard.decoded
    stop
  end
  anim maria point
  say maria decode.1
  say maria decode.2
  say kev-court decode.3
  say maria decode.4
  set decoded
  set goal 71
end

on look moved-board
  say maria chalkboard.decoded
end

on look relief
  if not board-moved
    say maria relief.hidden
    stop
  end
  say maria relief.look.1
  say maria relief.look.2
  if not stone-loose
    set goal 73
  end
end

on use relief
  if not board-moved
    say maria relief.blocked
    stop
  end
  if stone-loose
    say maria relief.heavy
  else
    say maria relief.use
  end
end

on use bonefolder relief
  if not board-moved
    say maria relief.blocked
    stop
  end
  if stone-loose
    say maria relief.heavy
    stop
  end
  anim maria reach
  sound paper
  set stone-loose
  say maria relief.loose.1
  say maria relief.loose.2
  set goal 74
end

on look loose-stone
  say maria loose-stone.look
end

on look hollow
  if took-token
    say maria hollow.empty
  else
    say maria hollow.look
  end
end

on use hollow
  if took-token
    say maria hollow.empty
    stop
  end
  anim maria reach
  sound coin
  gain token
  set took-token
  say maria token.found.1
  say maria token.found.2
  say maria token.found.3
  say kev-court token.found.4
  say maria token.found.5
  clear kept-phone
  if has phone
    set kept-phone
    lose phone
  end
  clear kept-page
  if has page
    set kept-page
    lose page
  end
  clear kept-bonefolder
  if has bonefolder
    set kept-bonefolder
    lose bonefolder
  end
  clear kept-brush
  if has brush
    set kept-brush
    lose brush
  end
  clear kept-napkin
  if has napkin
    set kept-napkin
    lose napkin
  end
  clear kept-token
  if has token
    set kept-token
    lose token
  end
  goto quay start
end

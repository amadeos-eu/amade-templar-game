# Act 2, scene 6: the Cour du Temple at night, where the Templars of La Rochelle had their house.

on enter court
  if fog-seen
    if not act2-done
      set act2-done
    if kept-phone
      gain phone
    end
    if kept-page
      gain page
    end
    if kept-bonefolder
      gain bonefolder
    end
    if kept-brush
      gain brush
    end
    if kept-napkin
      gain napkin
    end
    if kept-token
      gain token
    end
      say lena court.end.1
      say lena court.end.2
      say lena court.end.3
      anim kev-court point
      say kev-court court.end.4
      say lena court.end.5
      say kev-court court.end.6
      say lena court.end.7
      title card.act2.end
      goto charola arrival
    end
    stop
  end
  if not been-court
    set been-court
    say lena court.intro.1
    say kev-court court.intro.2
    say lena court.intro.3
    set goal 70
  end
end

on look night-sky
  say lena night-sky.look
end

on look old-wall
  say lena old-wall.look
end

on use old-wall
  say lena old-wall.use
end

on look street-sign
  say lena street-sign.look
end

on look cafe
  say lena cafe.look
end

on use cafe
  say lena cafe.use
end

on look cafe-window
  say lena cafe-window.look
end

on look terrace-lights
  say lena terrace-lights.look
end

on exit back-harbour
  if act2-done
    stop
  end
  goto harbour old-town
end

# ---- the menu board and the stone behind it ----

on look chalkboard
  if decoded
    say lena chalkboard.decoded
  else
    say lena chalkboard.look.1
    say lena chalkboard.look.2
  end
  say lena chalkboard.behind
end

on use chalkboard
  if not allowed
    say lena chalkboard.ask
    stop
  end
  anim lena reach
  sound door
  hide chalkboard
  show moved-board
  set board-moved
  say lena chalkboard.moved
  set goal 72
end

on use napkin chalkboard
  if decoded
    say lena chalkboard.decoded
    stop
  end
  anim lena point
  say lena decode.1
  say lena decode.2
  say kev-court decode.3
  say lena decode.4
  set decoded
  set goal 71
end

on look moved-board
  say lena chalkboard.decoded
end

on look relief
  if not board-moved
    say lena relief.hidden
    stop
  end
  say lena relief.look.1
  say lena relief.look.2
  if not stone-loose
    set goal 73
  end
end

on use relief
  if not board-moved
    say lena relief.blocked
    stop
  end
  if stone-loose
    say lena relief.heavy
  else
    say lena relief.use
  end
end

on use bonefolder relief
  if not board-moved
    say lena relief.blocked
    stop
  end
  if stone-loose
    say lena relief.heavy
    stop
  end
  anim lena reach
  sound paper
  set stone-loose
  say lena relief.loose.1
  say lena relief.loose.2
  set goal 74
end

on look loose-stone
  say lena loose-stone.look
end

on look hollow
  if took-token
    say lena hollow.empty
  else
    say lena hollow.look
  end
end

on use hollow
  if took-token
    say lena hollow.empty
    stop
  end
  anim lena reach
  sound coin
  gain token
  set took-token
  say lena token.found.1
  say lena token.found.2
  say lena token.found.3
  say kev-court token.found.4
  say lena token.found.5
  clear kept-phone
  if has phone
    set kept-phone
    lose phone
  end
  clear kept-page
  if has page
    set kept-page
    lose page
  end
  clear kept-bonefolder
  if has bonefolder
    set kept-bonefolder
    lose bonefolder
  end
  clear kept-brush
  if has brush
    set kept-brush
    lose brush
  end
  clear kept-napkin
  if has napkin
    set kept-napkin
    lose napkin
  end
  clear kept-token
  if has token
    set kept-token
    lose token
  end
  goto quay start
end

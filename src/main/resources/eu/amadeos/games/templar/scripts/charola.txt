# Act 3, scene 7: the Charola, the round church of the Templars at Tomar. Dona Inês, her showcase and her questions.

on enter charola
  if not been-charola
    set been-charola
    title card.tomar
    say lena charola.intro.1
    say helene-tomar charola.helene.1
    say lena charola.helene.2
    say helene-tomar charola.helene.3
    say lena charola.helene.4
    say helene-tomar charola.helene.5
    walk helene-tomar 70 308
    hide helene-tomar
    sound door
    say kev-tomar charola.kev.1
    say ines charola.kev.2
    say kev-tomar charola.kev.3
    set goal 90
  end
end

on look octagon

  say lena octagon.look.2
end

  say lena octagon.use
end

on look paintings
  say lena paintings.look
end

on use paintings

end

on look charola-window
  say lena charola-window.look
end

on look rope

on exit tower-door
  if not archive-open
    say lena tower-door.locked
    stop
  end
  goto archive arrival
end

# ---- the showcase ----

on look showcase
  if token-fits
    say lena showcase.after
    stop
  end
  say lena showcase.look.1
  say lena showcase.look.2
  if quiz-done
    say lena showcase.open
  else
    set may-ask
    set goal 91
  end
end

on use showcase
  if quiz-done
    say lena showcase.open
  else
    say lena showcase.locked
  end
end

on use token showcase
  if token-fits
    say lena showcase.after
    stop
  end
  if not quiz-done
    say lena showcase.glass
    stop
  end
  anim lena reach
  sound discovery
  set token-fits
  say lena token.fits.1
  say lena token.fits.2
  say ines token.fits.3
  say ines token.fits.4
  say lena token.fits.5
  say ines token.fits.6
  walk ines 90 316
  face ines left
  sound door
  set archive-open
  say ines token.fits.7
  set goal 94
end

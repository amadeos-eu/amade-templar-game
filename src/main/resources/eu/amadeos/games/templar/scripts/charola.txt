# Act 3, scene 7: the Charola, the round church of the Templars at Tomar. Dona Inês, her showcase and her questions.

on enter charola
  if not been-charola
    set been-charola
    title card.tomar
    say maria charola.intro.1
    say helene-tomar charola.helene.1
    say maria charola.helene.2
    say helene-tomar charola.helene.3
    say maria charola.helene.4
    say helene-tomar charola.helene.5
    walk helene-tomar 70 308
    hide helene-tomar
    sound door
    say kev-tomar charola.kev.1
    say ines charola.kev.2
    say kev-tomar charola.kev.3
    set goal 90
  end
end

on look octagon
  say maria octagon.look.1
  say maria octagon.look.2
end

on use octagon
  say maria octagon.use
end

on look paintings
  say maria paintings.look
end

on use paintings
  say maria paintings.use
end

on look charola-window
  say maria charola-window.look
end

on look rope
  say maria rope.look
end

on use rope
  say ines rope.use
end

on exit tower-door
  if not archive-open
    say maria tower-door.locked
    stop
  end
  goto archive arrival
end

# ---- the showcase ----

on look showcase
  if token-fits
    say maria showcase.after
    stop
  end
  say maria showcase.look.1
  say maria showcase.look.2
  if quiz-done
    say maria showcase.open
  else
    set may-ask
    set goal 91
  end
end

on use showcase
  if quiz-done
    say maria showcase.open
  else
    say maria showcase.locked
  end
end

on use token showcase
  if token-fits
    say maria showcase.after
    stop
  end
  if not quiz-done
    say maria showcase.glass
    stop
  end
  anim maria reach
  sound discovery
  set token-fits
  say maria token.fits.1
  say maria token.fits.2
  say ines token.fits.3
  say ines token.fits.4
  say maria token.fits.5
  say ines token.fits.6
  walk ines 90 316
  face ines left
  sound door
  set archive-open
  say ines token.fits.7
  set goal 94
end

# Three questions; a wrong answer starts them again, and costs nothing.

# --8<-- [start:quiz]
option quiz arrest-1314 quiz.a.1314 if quiz == 0
  say maria quiz.1314.1
  say ines quiz.1314.2
  set quiz 0
  exit
end

option quiz arrest-1307 quiz.a.1307 if quiz == 0
  say maria quiz.1307.1
  say ines quiz.right.1
  set quiz 1
  say ines quiz.q.2
end

option quiz arrest-1119 quiz.a.1119 if quiz == 0
  say maria quiz.1119.1
  say ines quiz.1119.2
  set quiz 0
  exit
end
